function buildUPRMilestones(n) {

  const start = 21, end = 56;

  const checklist = [

    "Filters and slicers show the right options and data",

    "No hidden exclusions in the filters",

    "Default filters open to the right view",

    "Cross-filtering and drill-through work correctly",

    "Row-level security shows only what each user should see",

    "Reset filters returns to the default view",

  ];

  const rounds = [];

  for (let i = 1; i <= n; i++) {

    const day = start + (i / n) * (end - start);

    const isLast = i === n;

    const title = isLast ? "Full Review (Feature-Complete)" : (n === 1 ? "Full Review (Feature-Complete)" : `UPR Review #${i}`);

    rounds.push({

      day,

      title,

      checklist,

      why: isLast

        ? "Every feature gets reviewed in full - this is the last easy point to make big changes."

        : "Creates a checkpoint mid-build so the team doesn't build too far on top of broken logic.",

      pmAction: isLast ? "Confirm scope is locked from here on - changes get expensive after this point." : null,

      note: "Includes a fix and re-review pass before moving to the next checkpoint.",

    });

  }

  return rounds;

}


function addDays(date, days) {

  const d = new Date(date);

  d.setDate(d.getDate() + Math.round(days));

  return d;

}

function fmtDate(d) { return d.toLocaleDateString(undefined, { month: "short", day: "numeric", year: "numeric" }); }

function fmtDow(d) { return d.toLocaleDateString(undefined, { weekday: "long" }); }

function icsDate(d) {

  const y = d.getFullYear(), m = String(d.getMonth()+1).padStart(2,"0"), day = String(d.getDate()).padStart(2,"0");

  return `${y}${m}${day}`;

}

function icsEscape(t) { return t.replace(/\\/g,"\\\\").replace(/;/g,"\\;").replace(/,/g,"\\,").replace(/\n/g,"\\n"); }

}


function generatePlan() {

  const startVal = document.getElementById("startDate").value;

  const weeksVal = parseFloat(document.getElementById("totalWeeks").value);

  const uprInput = document.getElementById("uprRounds").value;

  const uprRounds = uprInput ? Math.max(1, Math.min(6, parseInt(uprInput, 10))) : 2;


  if (!startVal) { alert("Pick a project start date first."); return; }

  if (!weeksVal || weeksVal < 2) { alert("Enter a total delivery window of at least 2 weeks."); return; }


  const startDate = new Date(startVal + "T00:00:00");

  const totalDays = weeksVal * 7;

  const scale = totalDays / REF_DAYS;


  const now = new Date();

  const todayMidnight = new Date(now.getFullYear(), now.getMonth(), now.getDate());

  const daysSinceStart = Math.round((todayMidnight - startDate) / 86400000);


  let progressMode;

  if (daysSinceStart < 0) progressMode = "future";

  else if (daysSinceStart > totalDays) progressMode = "complete";

  else progressMode = "inprogress";


  document.getElementById("resultWrap").style.display = "block";

  const endDate = addDays(startDate, totalDays);

  document.getElementById("resultHeading").textContent =

    `${weeksVal}-week plan · ${fmtDate(startDate)} → ${fmtDate(endDate)} · ${uprRounds} UPR round${uprRounds > 1 ? "s" : ""}`;


  const allMilestones = [

    ...FIXED_MILESTONES_EARLY,

    ...buildUPRMilestones(uprRounds),

    ...FIXED_MILESTONES_LATE,

  ].sort((a, b) => a.day - b.day);


  // Filter to what fits in the timeline and tag each with done/upcoming

  let num = 0;

  const filteredMilestones = [];

  allMilestones.forEach(m => {

    const scaledDay = m.day * scale;

    if (scaledDay > totalDays) return;

    num++;

    const done = progressMode === "complete" ? true : (progressMode === "future" ? false : scaledDay <= daysSinceStart);

    filteredMilestones.push({ ...m, num, scaledDay, done, phaseColor: getPhaseColorForDay(m.day) });

  });


  // ---- Ruler ----

  const ruler = document.getElementById("ruler");

  ruler.innerHTML = "";

  ruler.style.minWidth = Math.max(700, weeksVal * 66) + "px";


  const track = document.createElement("div");

  track.className = "phase-track";

  PHASES.forEach(ph => {

    const seg = document.createElement("div");

    seg.className = "phase-seg";

    seg.style.width = (((ph.end - ph.start) / REF_DAYS) * 100) + "%";

    seg.style.background = ph.color;

    const lbl = document.createElement("div");

    lbl.className = "phase-label";

    lbl.textContent = ph.name;

    seg.appendChild(lbl);

    track.appendChild(seg);

  });

  ruler.appendChild(track);


  const tickline = document.createElement("div");

  tickline.className = "tickline";

  ruler.appendChild(tickline);


  const weekCount = Math.round(weeksVal);

  for (let w = 0; w <= weekCount; w++) {

    const posPct = (w * 7 / totalDays) * 100;

    if (posPct > 100.5) continue;

    const tick = document.createElement("div");

    tick.className = "tick"; tick.style.left = posPct + "%";

    ruler.appendChild(tick);

    const tlabel = document.createElement("div");

    tlabel.className = "tick-label";

    tlabel.style.left = posPct + "%";

    if (posPct <= 3) tlabel.style.transform = "translateX(0)";

    else if (posPct >= 97) tlabel.style.transform = "translateX(-100%)";

    tlabel.textContent = "W" + w;

    ruler.appendChild(tlabel);

  }


  if (progressMode === "inprogress") {

    const todayPct = (daysSinceStart / totalDays) * 100;

    const todayLine = document.createElement("div");

    todayLine.className = "today-marker";

    todayLine.style.left = todayPct + "%";

    ruler.appendChild(todayLine);

    const todayLabel = document.createElement("div");

    todayLabel.className = "today-label";

    todayLabel.style.left = todayPct + "%";

    if (todayPct <= 5) todayLabel.style.transform = "translateX(0)";

    else if (todayPct >= 95) todayLabel.style.transform = "translateX(-100%)";

    todayLabel.textContent = "Today";

    ruler.appendChild(todayLabel);

  }


  const legend = document.getElementById("legend");

  legend.innerHTML = "";

  filteredMilestones.forEach(m => {

    const posPct = (m.scaledDay / totalDays) * 100;

    const dot = document.createElement("div");

    dot.className = "flag-dot" + (m.done ? " done" : "");

    dot.style.left = posPct + "%"; dot.textContent = m.num; dot.title = m.title;

    ruler.appendChild(dot);

    const li = document.createElement("div");

    li.className = "legend-item" + (m.done ? " done" : "");

    li.innerHTML = `<span class="legend-num">${m.done ? "✓" : m.num}</span>${m.title}`;

    legend.appendChild(li);

  });


  // ---- Stepper ----

  const stepper = document.getElementById("stepper");

  stepper.innerHTML = "";

  const stepperLine = document.createElement("div");

  stepperLine.className = "stepper-line";

  stepper.appendChild(stepperLine);

  filteredMilestones.forEach((m, idx) => {

    const pct = filteredMilestones.length > 1 ? (idx / (filteredMilestones.length - 1)) * 96 + 2 : 50;

    const node = document.createElement("div");

    node.className = "stepper-node " + (m.done ? "done" : "upcoming");

    node.style.left = pct + "%";

    node.textContent = m.done ? "✓" : m.num;

    node.title = m.title;

    stepper.appendChild(node);

  });


  // ---- Next up banner ----

  const nextUpEl = document.getElementById("nextUpBanner");

  if (progressMode === "complete") {

    const last = filteredMilestones[filteredMilestones.length - 1];

    const d = addDays(startDate, last.scaledDay);

    nextUpEl.innerHTML = `<strong>Project complete.</strong> Last checkpoint was ${last.title} on ${fmtDate(d)}.`;

  } else if (progressMode === "future") {

    const daysToStart = -daysSinceStart;

    const first = filteredMilestones[0];

    const d = addDays(startDate, first.scaledDay);

    nextUpEl.innerHTML = `<strong>Starts in ${daysToStart} day${daysToStart === 1 ? "" : "s"}.</strong> First checkpoint: ${first.title} on ${fmtDate(d)}.`;

  } else {

    const next = filteredMilestones.find(m => !m.done);

    if (next) {

      const d = addDays(startDate, next.scaledDay);

      const daysUntil = Math.ceil((d - todayMidnight) / 86400000);

      const when = daysUntil <= 0 ? "today" : `in ${daysUntil} day${daysUntil === 1 ? "" : "s"}`;

      nextUpEl.innerHTML = `<strong>Next up:</strong> ${next.title} — ${when} (${fmtDate(d)}).`;

    } else {

      nextUpEl.innerHTML = `<strong>All checkpoints done.</strong> Nice work.`;

    }

  }


  // ---- RACI table ----

  const raciBody = document.getElementById("raciBody");

  raciBody.innerHTML = "";

  PHASES.forEach(ph => {

    const s = addDays(startDate, ph.start * scale);

    const e = addDays(startDate, ph.end * scale);

    const tr = document.createElement("tr");

    tr.innerHTML = `

      <td class="phase-cell" data-label="Phase">${ph.name}<span class="dates">${fmtDate(s)} – ${fmtDate(e)}</span></td>

      <td class="col-dev" data-label="Developer">${ph.dev}</td>

      <td class="col-qa" data-label="QA">${ph.qa}</td>

      <td class="col-pm" data-label="PM">${ph.pm}</td>`;

    raciBody.appendChild(tr);

  });


  // ---- Checkpoint action list ----

  const actionList = document.getElementById("actionList");

  actionList.innerHTML = "";

  currentICSEvents = [];

  filteredMilestones.forEach(m => {

    const d = addDays(startDate, m.scaledDay);

    const needsAction = !!m.pmAction;

    let badgeClass, badgeText;

    if (m.done) { badgeClass = "done"; badgeText = "Completed"; }

    else if (needsAction) { badgeClass = "needed"; badgeText = "PM Action"; }

    else { badgeClass = "monitor"; badgeText = "Monitor Only"; }


    const checklistHTML = `<p class="checklist-label">Checks</p><ul class="checklist-ul">${

      m.checklist.map(c => `<li style="--dotcolor:${m.phaseColor}">${c}</li>`).join("")

    }</ul>`;


    const card = document.createElement("div");

    card.className = "action-card " + (m.done ? "done" : (needsAction ? "needed" : "monitor"));

    card.innerHTML = `

      <div class="ac-date">${fmtDate(d)}<span class="dow">${fmtDow(d)}</span></div>

      <div class="ac-body">

        <h3><span class="ac-num">${m.num}</span>${m.title} <span class="badge ${badgeClass}">${badgeText}</span></h3>

        <p>${m.why}</p>

        ${checklistHTML}

        ${m.note ? `<p class="note-line">${m.note}</p>` : ``}

        ${(!m.done && needsAction) ? `<p class="pm-line">→ ${m.pmAction}</p>` : ``}

      </div>`;

    actionList.appendChild(card);


    let desc = m.checklist.join("; ") + ". " + m.why;

    desc += needsAction ? " PM action: " + m.pmAction : " No PM action needed — monitor only.";

    currentICSEvents.push({ date: d, title: m.title, description: desc });

  });


  // ---- Reference tab: triggers ----

  const triggerList = document.getElementById("triggerList");

  triggerList.innerHTML = "";

  SUBMISSION_TRIGGERS.forEach(t => {

    const item = document.createElement("div");

    item.className = "trigger-item";

    item.innerHTML = `<span class="dot"></span><span>${t}</span>`;

    triggerList.appendChild(item);

  });

}


function downloadICS() {

  if (!currentICSEvents.length) { alert("Generate a plan first."); return; }

  let ics = "BEGIN:VCALENDAR\r\nVERSION:2.0\r\nPRODID:-//EVOCS Dashboard Delivery Planner//EN\r\nCALSCALE:GREGORIAN\r\n";

  const stamp = icsDate(new Date()) + "T000000Z";

  currentICSEvents.forEach((ev, i) => {

    const dt = icsDate(ev.date);

    ics += "BEGIN:VEVENT\r\n";

    ics += `UID:evocs-planner-${dt}-${i}@dashboard-project\r\n`;

    ics += `DTSTAMP:${stamp}\r\n`;

    ics += `DTSTART;VALUE=DATE:${dt}\r\n`;

    ics += `SUMMARY:${icsEscape("Dashboard Project: " + ev.title)}\r\n`;

    ics += `DESCRIPTION:${icsEscape(ev.description)}\r\n`;

    ics += "END:VEVENT\r\n";

  });

  ics += "END:VCALENDAR\r\n";

  const blob = new Blob([ics], { type: "text/calendar;charset=utf-8" });

  const url = URL.createObjectURL(blob);

  const a = document.createElement("a");

  a.href = url; a.download = "evocs_dashboard_checkpoints.ics";

  document.body.appendChild(a); a.click(); document.body.removeChild(a);

  URL.revokeObjectURL(url);

}


async function downloadPDF() {

  const pdfBtn = document.getElementById("pdfBtn");

  const resultWrap = document.getElementById("resultWrap");

  const panels = document.querySelectorAll(".tabpanel");

  const ruler = document.getElementById("ruler");


  if (typeof html2canvas === "undefined" || typeof window.jspdf === "undefined") {

    alert("Couldn't load the PDF tools. Check your internet connection and try again.");

    return;

  }


  const originalText = pdfBtn.textContent;

  pdfBtn.textContent = "Building PDF…";

  pdfBtn.disabled = true;


  // Show every tab's content, not just the active one

  const activeStates = [];

  panels.forEach(p => { activeStates.push(p.classList.contains("active")); p.style.display = "block"; });


  // Hide buttons that shouldn't show up in the PDF

  const hideEls = [pdfBtn, document.getElementById("icsBtnTop"), document.getElementById("icsBtn"), document.querySelector(".ics-text")].filter(Boolean);

  const prevDisplays = hideEls.map(el => el.style.display);

  hideEls.forEach(el => el.style.display = "none");


  // Shrink the timeline ruler to fit the page width instead of its normal scroll width.

  // All the marks on it are positioned by percentage, so this just rescales it cleanly.

  const prevRulerWidth = ruler.style.width;

  const prevRulerMinWidth = ruler.style.minWidth;

  ruler.style.minWidth = "0";

  ruler.style.width = "100%";


  // Give the browser a moment to apply the layout changes before measuring anything

  await new Promise(resolve => setTimeout(resolve, 50));


  try {

    // Elements that should never be sliced across a page break

    const atomicSelectors = [

      ".next-up", ".ruler-scroll", ".legend", ".stepper-wrap",

      ".raci-table tr", ".action-card", ".principle-box", ".watch-item",

      ".trigger-item", ".risk-note",

    ]; // h2.section-title is handled separately below, glued to whatever follows it

    const wrapRect = resultWrap.getBoundingClientRect();

    const totalHeightCSS = resultWrap.scrollHeight;

    const boundaries = new Set([0, totalHeightCSS]);


    function addBoundary(pxFromViewportTop) {

      const rel = Math.round(pxFromViewportTop - wrapRect.top);

      if (rel >= 0 && rel <= totalHeightCSS) boundaries.add(rel);

    }


    document.querySelectorAll(atomicSelectors.join(",")).forEach(el => {

      const r = el.getBoundingClientRect();

      addBoundary(r.top);

      addBoundary(r.bottom);

    });


    // A heading and whatever immediately follows it stay together —

    // no cutting a page right after a title with its content stranded on the next page

    document.querySelectorAll("h2.section-title").forEach(h => {

      const hRect = h.getBoundingClientRect();

      const sibling = h.nextElementSibling;

      const groupBottom = sibling ? sibling.getBoundingClientRect().bottom : hRect.bottom;

      addBoundary(hRect.top);

      addBoundary(groupBottom);

    });


    const sortedBoundaries = Array.from(boundaries).sort((a, b) => a - b);


    const canvas = await html2canvas(resultWrap, { scale: 2, backgroundColor: "#ffffff", useCORS: true });

    const scaleFactor = canvas.height / totalHeightCSS; // canvas px per CSS px


    const pdf = new window.jspdf.jsPDF("p", "pt", "a4");

    const pageWidthPt = pdf.internal.pageSize.getWidth();

    const pageHeightPt = pdf.internal.pageSize.getHeight();

    const ptPerCssPx = pageWidthPt / wrapRect.width;

    const cssPerPage = pageHeightPt / ptPerCssPx;


    // Find safe cut points — never inside an atomic element

    const cuts = [0];

    let current = 0;

    let guard = 0;

    while (current < totalHeightCSS - 0.5 && guard < 500) {

      guard++;

      const target = current + cssPerPage;

      if (target >= totalHeightCSS) { cuts.push(totalHeightCSS); break; }

      let candidate = null;

      for (const b of sortedBoundaries) {

        if (b > current && b <= target) candidate = b;

        if (b > target) break;

      }

      if (candidate === null) candidate = target; // a single element taller than a page — unavoidable

      cuts.push(candidate);

      current = candidate;

    }


    for (let i = 0; i < cuts.length - 1; i++) {

      const sliceTopCss = cuts[i];

      const sliceBottomCss = cuts[i + 1];

      const sliceTopPx = Math.round(sliceTopCss * scaleFactor);

      const sliceHeightPx = Math.round((sliceBottomCss - sliceTopCss) * scaleFactor);

      if (sliceHeightPx <= 0) continue;


      const pageCanvas = document.createElement("canvas");

      pageCanvas.width = canvas.width;

      pageCanvas.height = sliceHeightPx;

      pageCanvas.getContext("2d").drawImage(

        canvas, 0, sliceTopPx, canvas.width, sliceHeightPx, 0, 0, canvas.width, sliceHeightPx

      );


      const pageImgData = pageCanvas.toDataURL("image/png");

      const sliceHeightPt = (sliceHeightPx / scaleFactor) * ptPerCssPx;


      if (i > 0) pdf.addPage();

      pdf.addImage(pageImgData, "PNG", 0, 0, pageWidthPt, sliceHeightPt);

    }


    pdf.save("evocs_dashboard_plan.pdf");

  } catch (err) {

    console.error(err);

    alert("Something went wrong building the PDF. Please try again.");

  } finally {

    panels.forEach((p, i) => { p.style.display = activeStates[i] ? "block" : "none"; });

    hideEls.forEach((el, i) => { el.style.display = prevDisplays[i]; });

    ruler.style.width = prevRulerWidth;

    ruler.style.minWidth = prevRulerMinWidth;

    pdfBtn.textContent = originalText;

    pdfBtn.disabled = false;

  }

}


window.addEventListener("DOMContentLoaded", () => {

  document.getElementById("startDate").value = new Date().toISOString().split("T")[0];

  generatePlan();

});
